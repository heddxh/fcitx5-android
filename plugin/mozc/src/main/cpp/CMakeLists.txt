cmake_minimum_required(VERSION 3.18)

project(fcitx5-android-plugin-mozc VERSION ${VERSION_NAME})

# For reproducible build
add_link_options("LINKER:--hash-style=gnu,--build-id=none")

# prefab dependency
find_package(fcitx5 REQUIRED CONFIG)
get_target_property(FCITX5_CMAKE_MODULES fcitx5::cmake INTERFACE_INCLUDE_DIRECTORIES)
set(CMAKE_MODULE_PATH ${FCITX5_CMAKE_MODULES} ${CMAKE_MODULE_PATH})

find_package(ECM MODULE)
find_package(Fcitx5Core MODULE)
find_package(Fcitx5Module MODULE)

set(FCITX5_MOZC_DIR "${CMAKE_CURRENT_SOURCE_DIR}/fcitx5-mozc")
# gradle module build dir; shared between ABIs and wiped by clean
set(MOZC_CACHE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../build/mozc")

if(EXISTS "${PREBUILT_DIR}/libmozc/${ANDROID_ABI}/lib/libmozc-static.a")
    # use prebuilt mozc when available
    add_library(Mozc_static STATIC IMPORTED)
    set_target_properties(Mozc_static PROPERTIES
            IMPORTED_LOCATION "${PREBUILT_DIR}/libmozc/${ANDROID_ABI}/lib/libmozc-static.a"
            INTERFACE_INCLUDE_DIRECTORIES "${PREBUILT_DIR}/libmozc/${ANDROID_ABI}/include/mozc"
            INTERFACE_LINK_LIBRARIES "-llog -L${PREBUILT_DIR}/libmozc/${ANDROID_ABI}/lib -labsl -lprotobuf -lutf8_validity"
            )
    set(MOZC_TARGET Mozc_static)
else()
    # build mozc from source
    if(NOT EXISTS "${FCITX5_MOZC_DIR}/mozc/src/BUILD.bazel")
        message(FATAL_ERROR "mozc sources not found. Fetch them with:\n"
                "git submodule update --init --recursive --checkout plugin/mozc/src/main/cpp/fcitx5-mozc")
    endif()

    # serialize configuration across ABIs: patching, host protoc build and
    # dictionary download are shared between them
    file(MAKE_DIRECTORY "${MOZC_CACHE_DIR}")
    file(LOCK "${MOZC_CACHE_DIR}/.configure.lock" GUARD PROCESS)

    # libprotoc links against libupb, which fcitx5-mozc disables
    execute_process(
            COMMAND git apply --reverse --check "${FCITX5_MOZC_DIR}/patches/protobuf.patch"
            WORKING_DIRECTORY "${FCITX5_MOZC_DIR}/protobuf"
            RESULT_VARIABLE PROTOBUF_PATCH_APPLIED
            OUTPUT_QUIET
            ERROR_QUIET
    )
    if(NOT PROTOBUF_PATCH_APPLIED EQUAL 0)
        execute_process(
                COMMAND git apply "${FCITX5_MOZC_DIR}/patches/protobuf.patch"
                WORKING_DIRECTORY "${FCITX5_MOZC_DIR}/protobuf"
                RESULT_VARIABLE PROTOBUF_PATCH_RESULT
        )
        if(NOT PROTOBUF_PATCH_RESULT EQUAL 0)
            message(FATAL_ERROR "Cannot apply fcitx5-mozc/patches/protobuf.patch")
        endif()
    endif()

    # protoc has to run on the build host during cross compilation
    if(NOT PROTOC_EXECUTABLE)
        set(PROTOC_EXECUTABLE "${MOZC_CACHE_DIR}/host/protobuf/protoc")
        if(NOT EXISTS "${PROTOC_EXECUTABLE}")
            message(STATUS "Building protoc for the build host")
            execute_process(
                    COMMAND ${CMAKE_COMMAND} -S "${FCITX5_MOZC_DIR}" -B "${MOZC_CACHE_DIR}/host"
                            -G "${CMAKE_GENERATOR}" "-DCMAKE_MAKE_PROGRAM=${CMAKE_MAKE_PROGRAM}"
                            -DCMAKE_BUILD_TYPE=Release -DBUILD_MOZC_ADDON=OFF
                    RESULT_VARIABLE HOST_PROTOC_CONFIGURE_RESULT
            )
            if(NOT HOST_PROTOC_CONFIGURE_RESULT EQUAL 0)
                message(FATAL_ERROR "Cannot configure fcitx5-mozc for the build host")
            endif()
            execute_process(
                    COMMAND ${CMAKE_COMMAND} --build "${MOZC_CACHE_DIR}/host" --target protoc --parallel
                    RESULT_VARIABLE HOST_PROTOC_BUILD_RESULT
            )
            if(NOT HOST_PROTOC_BUILD_RESULT EQUAL 0)
                message(FATAL_ERROR "Cannot build protoc for the build host")
            endif()
        endif()
    endif()

    # dictionary data generated by https://github.com/fcitx-contrib/fcitx5-mozc CI,
    # gets embedded into libmozc.so
    set(MOZC_DATA_INC "${MOZC_CACHE_DIR}/mozc_data.inc")
    if(NOT EXISTS "${MOZC_DATA_INC}")
        message(STATUS "Downloading mozc_data.inc")
        file(DOWNLOAD
                "https://github.com/fcitx-contrib/fcitx5-mozc/releases/download/latest/mozc_data.inc"
                "${MOZC_DATA_INC}.part"
                STATUS MOZC_DATA_DOWNLOAD_STATUS
        )
        list(GET MOZC_DATA_DOWNLOAD_STATUS 0 MOZC_DATA_DOWNLOAD_ERROR)
        if(NOT MOZC_DATA_DOWNLOAD_ERROR EQUAL 0)
            file(REMOVE "${MOZC_DATA_INC}.part")
            message(FATAL_ERROR "Cannot download mozc_data.inc. "
                    "Download it manually and place it at ${MOZC_DATA_INC}")
        endif()
        file(RENAME "${MOZC_DATA_INC}.part" "${MOZC_DATA_INC}")
    endif()
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/fcitx5-mozc/data_manager/oss")
    file(CREATE_LINK "${MOZC_DATA_INC}"
            "${CMAKE_CURRENT_BINARY_DIR}/fcitx5-mozc/data_manager/oss/mozc_data.inc"
            COPY_ON_ERROR SYMBOLIC
    )

    # PROTOC_EXECUTABLE is provided above
    set(protobuf_BUILD_PROTOC_BINARIES OFF)
endif()

add_subdirectory(fcitx5-mozc)

target_include_directories(mozc PRIVATE
        "${PROJECT_SOURCE_DIR}/../../../../../lib/fcitx5/build/headers/usr/include"
        "${PROJECT_SOURCE_DIR}/../../../../../lib/fcitx5/build/headers/usr/include/Fcitx5/Module"
        )

target_link_libraries(mozc log)
